// These are source file name and line number tokens inserted by the
// SrcReader class file reader when switching to or returning from a new
// source file. Since file names and line numbers change each time we enter
// or leave a source file these must be updated and this is done via
// these inserted tokens. The tokens are thus skipped in parsing, after
// extracting the file information.
//SKIP :
//{
//     {ThreePL.setPosition(image);}
//}

// Update the file line number on each newline, but skip the token.

<DEFAULT> SKIP : {
"\n" : {
}

   
// Skip carriage returns from DOS sources.

<DEFAULT> SKIP : {
"\r"
}

   
//SKIP :
//{
//        
//    |   )* "\n" >
//}

<DEFAULT> SKIP : {
<WHITE_SPACE: " " | "\t">
}

   
// Comments

<DEFAULT> SKIP : {
"//" : IN_SINGLE_LINE_COMMENT
| "/*" : IN_MULTI_LINE_COMMENT
}

   
<IN_SINGLE_LINE_COMMENT> SKIP : {
"\n" : DEFAULT
}

   
<IN_MULTI_LINE_COMMENT> SKIP : {
"*/" : DEFAULT
| "\n" : {
}

   
<IN_SINGLE_LINE_COMMENT,IN_MULTI_LINE_COMMENT> SKIP : {
<~[]>
}

NON-TERMINALS

// Module, procedure and function declarations are compiled into
// 'Module", 'Procedure' and 'Function' classes which extend
// class 'Body'.
// These are entered into HashMaps 'modules', 'procedures'
// and 'functions' in the enclosing module, procedure or function body.
// The code body for each of these is compiled into
// an instance of the 'Block' as a field of the super class 'Body'.
//
// Each tree node contains the location of the most appropriate
// associated token in the form of a SrcLoc instance. This is
// primarily for use in execution error messages. This
// information is usually passed to the constructor by passing
// the token as the final parameter.
//

/**
 * File module grammar production.
 * A scope for this file module, fmscope, is created here and passed to all
 * modules, procedures and functions declared within this file module. It is
 * a static scope. Variables created within this file module can be accessed
 * from within any modules, procedures and functions declared within it even
 * though this file module scope may no longer be on the call stack.
 * @return  module/procedure call node
 * @throws  ParseException on a parsing error
 */
Filemodule ::= <FMODSTART> ( ( Statement ) )* ( <FMODFINISH> | <EOF> ) 
/**
 * File source grammar production.
 * @param   body is the Body class to contain the code body
 * @param   fmscope is the current file module scope
 * @return  source statement node
 * @throws  ParseException on a parsing error
 */
Filesource ::= <FSRCSTART> ( ( Statement ) )* ( <FSRCFINISH> | <EOF> ) 
/**
 * Class, module, procedure or function call grammar production.
 * @param   stat_call is true if called as a statement rather than in
 *          the RHS of as assignment statement
 * @param   lhs is true if called on the LHS of a statement
 * @return  function definition node
 * @throws  ParseException on a parsing error
 */
Call ::= ( DereferencedVar | <CLASS> ) <LPAREN> ArgList <RPAREN> ( Subscript | Field )* ( ( <RPOINT> | <RRPOINT> ) ( Subscript | Field )* )* ( <INCR> | <DECR> )? 
/**
 * Procedure call in parameter list or 'for' statement initialisation grammar
 * production.
 * @return  procedure call node
 * @throws  ParseException on a parsing error
 */
ParProcCall ::= ( DereferencedVar | <CLASS> ) <LPAREN> ArgList <RPAREN> 
/**
 * A module, procedure or function call argument list grammar production.
 * @param   call is the call node
 * @param   arrow indicates if a <- is allowed as an argument separator
 * @throws  ParseException on a parsing error
 */
ArgList ::= ( ( Keymatch | Expr | CompVal ) )? ( ( <COMMA> | <LPOINT> ) ( ( Keymatch | Expr | CompVal ) )? )* 
/**
 * Keymatch (name=value) argument grammer production
 * @return  a keymatch node
 * @throws ParseException on a parsing error
 */
Keymatch ::= <IDENTIFIER> <EQ> ( Expr | CompVal ) 
/**
 * Common block definition grammar production.
 * @param   body is the module code body
 * @param   fmscope is the current file module scope
 * @throws  ParseException on a parsing error
 */
CommonBlock ::= <COMMON> Block 
/**
 * Group (3PL class) definition grammar production.
 * @param   body is the module code body
 * @param   fmscope is the current file module scope
 * @throws  ParseException on a parsing error
 */
GroupDef ::= ( <GLOBAL> )? ( <CLASS> <IDENTIFIER> <LPAREN> ParamList <RPAREN> ) Block 
/**
 * Module definition grammar production.
 * @param   body is the module code body
 * @param   fmscope is the current file module scope
 * @throws  ParseException on a parsing error
 */
ModDef ::= ( <GLOBAL> | <PRIVATE> )? ( <MODULE> <IDENTIFIER> <LPAREN> ParamList <RPAREN> | <LEADERMODULE> <LPAREN> <STRING_LITERAL> ( <COMMA> <INTEGER_LITERAL> )? <RPAREN> | <TRAILERMODULE> <LPAREN> <STRING_LITERAL> ( <COMMA> <INTEGER_LITERAL> )? <RPAREN> | <POSTMODULE> <LPAREN> <STRING_LITERAL> ( <COMMA> <INTEGER_LITERAL> )? <RPAREN> ) Block 
/**
 * Procedure definition grammar production.
 * @param   body is the procedure code body
 * @param   fmscope is the current file module scope
 * @throws  ParseException on a parsing error
 */
ProcDef ::= ( <GLOBAL> | <PRIVATE> )? <PROCEDURE> <IDENTIFIER> <LPAREN> ParamList <RPAREN> Block 
/**
 * Function definition grammar production.
 * @param   body is the function code body
 * @param   fmscope is the current file module scope
 * @throws  ParseException on a parsing error
 */
FuncDef ::= ( <GLOBAL> | <PRIVATE> )? <FUNCTION> <IDENTIFIER> <LPAREN> ParamList <RPAREN> Block 
/**
 * Module, procedure or function output or input parameters grammar production.
 * @param   body is the module, procedure or function
 * @throws  ParseException on a parsing error
 */
ParamList ::= ( ( ParProcCall | <VARARGS> ) ( ( <COMMA> | <LPOINT> ) ( ParProcCall | <VARARGS> )? )* )? 
/**
 * General code block grammar production.
 * @param   name is the module name
 * @param   btype is the block type
 * @param   body is the Module, Procedure or Function production or else null
 * @param   fmscope is the file module scope
 * @return  the code block root node
 * @throws  ParseException on a parsing error
 */
Block ::= <LBRACE> ( Statement )* <RBRACE> 
/**
 * Statement grammar production.
 * @param   body is the body containing the statement
 * @param   fmscope is the current file module scope
 * @return  the statement node
 * @throws  ParseException on a parsing error
 */
Statement ::= ( GroupDef | ModDef | ProcDef | FuncDef | CommonBlock | InlineModBlock | Call <SEMICOL> | ImFuncAssignStat <SEMICOL> | TargFuncAssignStat <SEMICOL> | ImAssignStat <SEMICOL> | TargAssignStat <SEMICOL> | NullPipeAssignStat <SEMICOL> | SeqWhileStat | SeqDoWhileStat | SeqBlock | ParWhileStat | ParDoWhileStat | ParBlock | SyncWhenStat | SyncWhileStat | SyncDoWhileStat | SyncBlock | WaitPriBlock | Filemodule | Filesource | WhenStat | BranchStat | IncrDecrStat <SEMICOL> | ReturnStat | BreakStat | ContinStat | IfStat | WhileStat | DoWhileStat | ForStat | SwitchStat | PetriNetStat | <SEMICOL> ) 
/**
 * Immediate assignment (=) grammar production.
 * @return  immediate assignment statement node
 * @throws  ParseException on a parsing error
 */
ImAssignStat ::= DereferencedVar ( <EQ> | <PLUSEQ> | <MINEQ> | <MULEQ> | <DIVEQ> | <REMEQ> | <BANDEQ> | <BOREQ> | <BXOREQ> | <LANDEQ> | <LOREQ> | <LXOREQ> | <LSEQ> | <RSEQ> ) ( Expr | CompVal ) 
/**
 * Immediate assignment (=) grammar production.
 * @return  immediate function assignment statement node
 * @throws  ParseException on a parsing error
 */
ImFuncAssignStat ::= Call ( <EQ> | <PLUSEQ> | <MINEQ> | <MULEQ> | <DIVEQ> | <REMEQ> | <BANDEQ> | <BOREQ> | <BXOREQ> | <LANDEQ> | <LOREQ> | <LXOREQ> | <LSEQ> | <RSEQ> ) ( Expr | CompVal ) 
/**
 * Target assignment { @literal (:=, |-, <- or <<-) } grammar production.
 * @return  target assignment statement node
 * @throws  ParseException on a parsing error
 */
TargAssignStat ::= DereferencedVar ( <COLEQ> | <BUSPOINT> | <LPOINT> | <PLUSLPOINT> | <MINUSLPOINT> | <MULLPOINT> | <LLPOINT> ) ( Expr | CompVal ) 
/**
 * Target assignment {@literal (:=, |-, <- or <<-) } grammar production.
 * @return  target function assignment statement node
 * @throws  ParseException on a parsing error
 */
TargFuncAssignStat ::= Call ( <COLEQ> | <BUSPOINT> | <LPOINT> | <PLUSLPOINT> | <MINUSLPOINT> | <MULLPOINT> | <LLPOINT> ) ( Expr | CompVal ) 
/**
 * A null pipe assignment (pop a pipe) grammar production.
 * @return  null pipe assignment statement node

 */
NullPipeAssignStat ::= <NULL> <LLPOINT> DereferencedVar
unless ::= <UNLESS> <LPAREN> Expr <RPAREN> Block

WaitPriBlock ::= ( <WAITPRI> | <WAITPRILOCK> ) <LPAREN> DereferencedVar <COMMA> Expr <RPAREN> Statement ( unless )? 
/**

 * @throws  ParseException on a parsing error
 */
Expr15 ::= ( <RANGLE> DereferencedVar | DereferencedVar ) 
/**
 * Compound value grammar production.
 * @return  compound value node
 * @throws  ParseException on a parsing error
 */
CompVal ::= <LBRACE> ( Keymatch | Expr | CompVal )? ( <COMMA> ( Keymatch | Expr | CompVal )? )* <RBRACE>
DereferencedVar ::= FlaggedVar ( ( <RPOINT> | <RRPOINT> ) ( Subscript | Field )* )* ( <INCR> | <DECR> )? 
/**
 * Pre or post increment or decrement or indirect variable grammar production.
 * ++, --, * and {@literal &} are not handled as normal operator nodes.
 * The VarNode to which they apply has flags set
 * to indicate ++v, --v, v++, v-- , *v or {@literal &v}.
 * These are handled when the variable is accessed.
 * @return  a node for a variable with a ++ or --
 * @throws  ParseException on a parsing error
 */
FlaggedVar ::= ( <INCR> | <DECR> )? Variable 
/**
 * A variable (possibly with subscripts or field names) grammar production.
 * @return  a variable node
 * @throws  ParseException on a parsing error
 */
Variable ::= ( <GLOBALDOT> | <FILEDOT> | <CALLINGDOT> | <LOCALDOT> | <CLASSDOT> )? <IDENTIFIER> ( Subscript | Field )* 
/**
 * Subscript(s) grammar production.
 * @return  a subscript node
 * @throws  ParseException on a parse error
 */
Subscript ::= <LBRACK> ( ( Expr ( <DOTDOT> Expr )? )? ) <RBRACK> 
/**
 * Struct field (on a variable) grammar production.
 * @return  a field node
 * @throws  ParseException on a parse error
 */
Field ::= <DOT> <IDENTIFIER>
